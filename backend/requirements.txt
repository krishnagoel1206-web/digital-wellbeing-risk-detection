# Backend dependencies for the Digital Wellbeing Risk & Deviation Detection API.
#
# Versions are PINNED exactly, for two reasons:
#  1. Reproducibility. The brief requires model results to be reproducible, and
#     an unpinned scikit-learn or numpy can silently change numeric output
#     between installs, which would make the reported metrics unreproducible.
#  2. Python 3.15 is very new. These specific versions were confirmed to have
#     prebuilt cp315 Windows wheels, so no C compiler is needed.
#
# Verified interpreter: CPython 3.15.1 (win_amd64).

# --- Web framework -----------------------------------------------------------
fastapi==0.162.0
# NOTE: deliberately NOT uvicorn[standard].
# The "standard" extra depends on PyYAML, which has no cp315 wheel at all
# (verified: "No matching distribution found for pyyaml"), so installing it
# on Python 3.15 would require a C toolchain to build from source. PyYAML is
# only used by uvicorn to read YAML --log-config files, which we do not use.
# Instead we pull the two pieces of "standard" we actually want, both of
# which do ship 3.15-compatible wheels:
#   watchfiles -> enables --reload during development
#   httptools  -> faster HTTP parsing
#   websockets -> websocket protocol support
uvicorn==0.54.0
watchfiles==1.3.0
httptools==0.9.0
websockets==17.4
pydantic==2.14.2
pydantic-settings==2.13.0
# Required by FastAPI for multipart/form-data, i.e. CSV file upload.
python-multipart==0.0.20

# --- Database ----------------------------------------------------------------
# SQLite is accessed through SQLAlchemy's ORM so the data-access layer stays
# separate from the ML code and the API handlers.
SQLAlchemy==2.0.44

# --- Scientific / ML stack ---------------------------------------------------
numpy==2.5.4
pandas==3.1.1
scipy==1.18.1
scikit-learn==1.9.1
# Model persistence: trained estimators are saved as .joblib artifacts so that
# training and inference stay strictly separate.
joblib==1.6.0

# --- Testing -----------------------------------------------------------------
pytest==9.0.1
# httpx is needed by fastapi.testclient / starlette's TestClient.
# 0.28.1 is the latest stable; the 1.0.devN releases are pre-release only.
httpx==0.28.1
